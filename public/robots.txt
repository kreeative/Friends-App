# Everything public is crawlable. The signed-in app needs a session and is not
# linked from anywhere a crawler can reach, but it is named here so nobody has
# to guess, and because a URL can be found without being linked: a shared
# message, a browser's history sync, a referrer header.
#
# Disallow is a request not to FETCH, not a promise not to index. A URL that
# is disallowed here can still appear in results, titleless, if Google finds
# it elsewhere. The second half of the job is the noindex tag the app sets on
# these same routes at runtime. See src/lib/seo.js.
User-agent: *
Allow: /

# The signed-in app.
Disallow: /signin
Disallow: /start
Disallow: /me
Disallow: /settings
Disallow: /money
Disallow: /goals
Disallow: /checkin
Disallow: /proofs
Disallow: /g/

# Two spellings of one library. /books is the canonical one and the only one
# in the sitemap; these render the same chapters and now say so in a canonical
# tag. Kept out of the crawl so the budget goes on the page that counts.
Disallow: /library
Disallow: /lectures

# The English alias for the studies. /etudes is the canonical one and the only
# one in the sitemap; /studies renders the same pages.
Disallow: /studies

# Tracking parameters make one page look like several.
Disallow: /*?utm_
